// InputRemap — substitui o Logi Options+ para o Lift e o MX Keys S.
//
// Mouse (Lift):
//   Botão do meio (rodinha) segurado + arrastar:
//     ← / →  troca de mesa (Ctrl+← / Ctrl+→)
//     ↑      Mission Control (Ctrl+↑)
//     ↓      App Exposé (Ctrl+↓)
//   Botão do meio clique simples: Mission Control
//   Back    → ⌘C
//   Forward → ⌘V
// Teclado (MX Keys S):
//   Tecla de print → ⌃⇧⌘4 (região para o clipboard)
//
// Rodar com --debug para registrar eventos em /tmp/inputremap.log.

import AppKit
import ApplicationServices

let debug = CommandLine.arguments.contains("--debug")
let logURL = URL(fileURLWithPath: "/tmp/inputremap.log")

func log(_ s: String) {
    guard debug else { return }
    let line = "\(Date()) \(s)\n"
    if let h = try? FileHandle(forWritingTo: logURL) {
        h.seekToEndOfFile(); h.write(line.data(using: .utf8)!); try? h.close()
    } else {
        try? line.write(to: logURL, atomically: true, encoding: .utf8)
    }
}

// MARK: - Configuração

let gestureThreshold: Double = 60          // pixels de arrasto até disparar o gesto
let invertHorizontal = false               // true = arrastar p/ esquerda vai p/ mesa da direita

// Botões (CGEvent buttonNumber): 2 = meio, 3 = back, 4 = forward
let middleButton: Int64 = 2
let backButton: Int64 = 3
let forwardButton: Int64 = 4

// Tecla de print do MX Keys S. Preenchida depois da descoberta (--debug).
// keyCode = tecla normal; systemKey = tecla "de sistema" (NX_SYSDEFINED data1 >> 16).
let screenshotKeyCode: Int64? = nil
let screenshotSystemKey: Int? = nil

// MARK: - Envio de atalhos

enum Key: CGKeyCode {
    case c = 8, v = 9, four = 21
    case left = 123, right = 124, down = 125, up = 126
}

let src = CGEventSource(stateID: .hidSystemState)

func send(_ key: Key, _ flags: CGEventFlags) {
    var f = flags
    if [.left, .right, .up, .down].contains(key) {
        f.insert(.maskSecondaryFn); f.insert(.maskNumericPad)
    }
    for down in [true, false] {
        guard let e = CGEvent(keyboardEventSource: src, virtualKey: key.rawValue, keyDown: down) else { continue }
        e.flags = f
        e.post(tap: .cghidEventTap)
    }
}

// MARK: - Estado do gesto

var gestureActive = false
var gestureFired = false
var dx: Double = 0, dy: Double = 0

func fireGesture() {
    gestureFired = true
    if abs(dx) > abs(dy) {
        var goLeft = dx < 0
        if invertHorizontal { goLeft.toggle() }
        log("gesto: mesa \(goLeft ? "esquerda" : "direita")")
        send(goLeft ? .left : .right, .maskControl)
    } else if dy < 0 {
        log("gesto: Mission Control")
        send(.up, .maskControl)
    } else {
        log("gesto: App Exposé")
        send(.down, .maskControl)
    }
}

// MARK: - Event tap

var tap: CFMachPort?

let callback: CGEventTapCallBack = { _, type, event, _ in
    if type == .tapDisabledByTimeout || type == .tapDisabledByUserInput {
        if let t = tap { CGEvent.tapEnable(tap: t, enable: true) }
        log("tap reativado")
        return Unmanaged.passUnretained(event)
    }

    switch type {
    case .otherMouseDown, .otherMouseUp, .otherMouseDragged:
        let button = event.getIntegerValueField(.mouseEventButtonNumber)
        if debug && type != .otherMouseDragged { log("mouse \(type == .otherMouseDown ? "down" : "up") botão=\(button)") }

        if button == middleButton {
            switch type {
            case .otherMouseDown:
                gestureActive = true; gestureFired = false; dx = 0; dy = 0
            case .otherMouseDragged:
                guard gestureActive else { break }
                dx += Double(event.getIntegerValueField(.mouseEventDeltaX))
                dy += Double(event.getIntegerValueField(.mouseEventDeltaY))
                if !gestureFired && (abs(dx) > gestureThreshold || abs(dy) > gestureThreshold) {
                    fireGesture()
                }
            default: // up
                if gestureActive && !gestureFired {
                    log("clique do meio: Mission Control")
                    send(.up, .maskControl)
                }
                gestureActive = false
            }
            return nil
        }
        if button == backButton || button == forwardButton {
            if type == .otherMouseDown {
                send(button == backButton ? .c : .v, .maskCommand)
            }
            return nil
        }

    case .keyDown, .keyUp:
        let code = event.getIntegerValueField(.keyboardEventKeycode)
        if debug && type == .keyDown { log("tecla keyCode=\(code) flags=\(String(event.flags.rawValue, radix: 16))") }
        if let sk = screenshotKeyCode, code == sk {
            if type == .keyDown { send(.four, [.maskCommand, .maskShift, .maskControl]) }
            return nil
        }

    default:
        // NX_SYSDEFINED (14): teclas de mídia/sistema
        if type.rawValue == 14, let ns = NSEvent(cgEvent: event), ns.subtype.rawValue == 8 {
            let data = ns.data1
            let key = (data & 0xFFFF0000) >> 16
            let isDown = ((data & 0xFF00) >> 8) == 0x0A
            if debug && isDown { log("sistema key=\(key)") }
            if let sk = screenshotSystemKey, key == sk {
                if isDown { send(.four, [.maskCommand, .maskShift, .maskControl]) }
                return nil
            }
        }
    }
    return Unmanaged.passUnretained(event)
}

// MARK: - Inicialização

let opts = [kAXTrustedCheckOptionPrompt.takeUnretainedValue() as String: true] as CFDictionary
if !AXIsProcessTrustedWithOptions(opts) {
    log("sem permissão de Acessibilidade; aguardando…")
    while !AXIsProcessTrusted() { Thread.sleep(forTimeInterval: 2) }
}

let mask: CGEventMask =
    (1 << CGEventType.otherMouseDown.rawValue) |
    (1 << CGEventType.otherMouseUp.rawValue) |
    (1 << CGEventType.otherMouseDragged.rawValue) |
    (1 << CGEventType.keyDown.rawValue) |
    (1 << CGEventType.keyUp.rawValue) |
    (1 << 14)

guard let t = CGEvent.tapCreate(tap: .cghidEventTap, place: .headInsertEventTap,
                                options: .defaultTap, eventsOfInterest: mask,
                                callback: callback, userInfo: nil) else {
    log("falha ao criar event tap (permissão?)")
    exit(1)
}
tap = t
let source = CFMachPortCreateRunLoopSource(kCFAllocatorDefault, t, 0)
CFRunLoopAddSource(CFRunLoopGetCurrent(), source, .commonModes)
CGEvent.tapEnable(tap: t, enable: true)
log("InputRemap rodando")
CFRunLoopRun()
